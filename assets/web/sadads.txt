        // Web Audio API ile Tok Ahşap Sesi
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function playWoodClick() {
            if (audioCtx.state === 'suspended') audioCtx.resume();
            const t = audioCtx.currentTime;
            
            // Kısa ve tok bir gürültü (Noise) oluştur
            const bufferSize = audioCtx.sampleRate * 0.05; // 50ms uzunluk
            const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
            const data = buffer.getChannelData(0);
            for (let i = 0; i < bufferSize; i++) {
                data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize / 5)); // Azalan ses
            }
            const noise = audioCtx.createBufferSource();
            noise.buffer = buffer;

            // Filtre ile sesi toklaştır (Lowpass)
            const filter = audioCtx.createBiquadFilter();
            filter.type = 'lowpass';
            filter.frequency.value = 1000;

            noise.connect(filter);
            filter.connect(audioCtx.destination);
            noise.start(t);
        }

                    if (parseInt(b.style.top) > 240) {
                        b.style.top = '-60px'; // En üste gönder
                    }
                });
            }, 100);

            if(state.settings.sound) playWoodClick();
            if(state.settings.vibration) navigator.vibrate(15);
        }

        // --- 8. ZİKİR SEÇİMİ ---
        function initZikirOptions() {
            const container = document.getElementById('zikirOptions');
            zikirList.forEach(zikir => {
                const div = document.createElement('div');
                div.className = 'zikir-option';
                div.onclick = () => selectZikir(zikir.id);
                div.innerHTML = `<div><div class="zikir-option-name">${zikir.name}</div><div style="color: var(--text-muted); font-size: 12px;">${zikir.meaning}</div></div><div class="zikir-option-arabic">${zikir.arabic}</div>`;
                container.appendChild(div);
            });
        }
        function showZikirSelector() { document.getElementById('zikirModalOverlay').classList.add('active'); document.getElementById('zikirSelectionModal').classList.add('active'); }
        function closeZikirSelector() { document.getElementById('zikirModalOverlay').classList.remove('active'); document.getElementById('zikirSelectionModal').classList.remove('active'); }
        function selectZikir(id) {
            state.zikir.current = id;
            document.getElementById('currentTasbihName').textContent = zikirList.find(z=>z.id===id).name;
            closeZikirSelector();
            saveState();
        }

        // --- 9. DİĞER FONKSİYONLAR ---
        function saveState() { localStorage.setItem('buglemData', JSON.stringify(state)); }
        function loadState() {
            const d = localStorage.getItem('buglemData');
            if(d) Object.assign(state, JSON.parse(d));
        }
        function navigateTo(page, el) {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            document.getElementById(`page-${page}`).classList.add('active');
            document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
            el.classList.add('active');
            if(page === 'zikir') initTasbihBeads();
        }
        function goToProfile() { navigateTo('profile', document.querySelectorAll('.nav-item')[3]); }
        function updateStats() {
            document.getElementById('totalZikir').textContent = state.zikir.count;
            document.getElementById('totalSadaka').textContent = state.sadaka.totalGiven;
            document.getElementById('sadakaCount').textContent = state.sadaka.remaining;
        }
        function watchAd() {
            if(state.sadaka.remaining > 0) {
                alert("Reklam izleniyor... (Simülasyon)");
                state.sadaka.remaining--;
                state.sadaka.totalGiven++;
                saveState();
                updateStats();
            } else { alert("Günlük hakkınız doldu."); }
        }
        function copyContent(type) {
            const text = type === 'ayah' ? document.getElementById('dailyAyahTurkish').textContent : document.getElementById('dailyHadithText').textContent;
            navigator.clipboard.writeText(text);
            alert("Kopyalandı!");
        }
        function toggleNotifications() { state.settings.notifications = !state.settings.notifications; document.getElementById('notifToggle').classList.toggle('active'); saveState(); }
        function toggleSound() { state.settings.sound = !state.settings.sound; document.getElementById('soundToggle').classList.toggle('active'); saveState(); }
        function toggleVibration() { state.settings.vibration = !state.settings.vibration; document.getElementById('vibrationToggle').classList.toggle('active'); saveState(); }
        function resetData() { if(confirm("Sıfırlamak istediğine emin misin?")) { localStorage.removeItem('buglemData'); location.reload(); } }
